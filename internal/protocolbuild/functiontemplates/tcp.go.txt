func packet(kind, device, value byte) []byte { /* 定义 packet 函数。 */
	return []byte{0xAA, kind, device, value, byte(0xAA + kind + device + value)} /* 返回当前处理结果。 */
} /* 结束当前表达式或代码块。 */

func check(data []byte) error { /* 定义 check 函数。 */
	if len(data) != 5 || data[0] != 0xAA || (data[1] != 1 && data[1] != 2) || data[2] == 0 { /* 判断条件并选择处理分支。 */
		return fmt.Errorf("报文长度、类型或设备号无效") /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	if data[4] != byte(data[0]+data[1]+data[2]+data[3]) { /* 判断条件并选择处理分支。 */
		return fmt.Errorf("校验和错误") /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	return nil /* 返回当前处理结果。 */
} /* 结束当前表达式或代码块。 */

func decode(data []byte, ctx Context) (Message, error) { /* 定义 decode 函数。 */
	if err := check(data); err != nil { /* 判断条件并选择处理分支。 */
		return Message{}, err /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	if data[1] == 2 { /* 判断条件并选择处理分支。 */
		return Message{MessageType: "COMMAND_REPLY", Event: map[string]any{"type": "ACK"}}, nil /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	return properties(map[string]any{"temperature": int(data[3])}), nil /* 返回当前处理结果。 */
} /* 结束当前表达式或代码块。 */

func ingress(data []byte, ctx Context) (Frame, error) { /* 定义 ingress 函数。 */
	if len(data) > 0 && data[0] != 0xAA { /* 判断条件并选择处理分支。 */
		return Frame{}, fmt.Errorf("帧头错误") /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	if len(data) < 5 { /* 判断条件并选择处理分支。 */
		return Frame{NeedMore: true}, nil /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	if err := check(data[:5]); err != nil { /* 判断条件并选择处理分支。 */
		return Frame{}, err /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	sequence, _ := ctx.State["sequence"].(float64)                                                                                          /* 更新 _ 的值。 */
	result := Frame{Consumed: 5, DeviceID: strconv.Itoa(int(data[2])), State: map[string]any{"device": int(data[2]), "sequence": sequence}} /* 更新 result 的值。 */
	if data[1] == 2 {                                                                                                                       /* 判断条件并选择处理分支。 */
		result.CorrelationID = strconv.Itoa(int(data[3])) /* 更新 result.CorrelationID 的值。 */
	} else { /* 结束当前表达式或代码块。 */
		result.Reply = packet(2, data[2], data[3]) /* 更新 result.Reply 的值。 */
	} /* 结束当前表达式或代码块。 */
	return result, nil /* 返回当前处理结果。 */
} /* 结束当前表达式或代码块。 */

func encode(command Command, ctx Context) (Frame, error) { /* 定义 encode 函数。 */
	if command.Type != "ping" { /* 判断条件并选择处理分支。 */
		return Frame{}, fmt.Errorf("不支持命令 %s", command.Type) /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	device, _ := ctx.State["device"].(float64) /* 更新 _ 的值。 */
	if device < 1 || device > 255 {            /* 判断条件并选择处理分支。 */
		return Frame{}, fmt.Errorf("请等待设备上报以取得地址") /* 返回当前处理结果。 */
	} /* 结束当前表达式或代码块。 */
	previous, _ := ctx.State["sequence"].(float64)                                                                                                                          /* 更新 _ 的值。 */
	sequence := int(previous)%255 + 1                                                                                                                                       /* 更新 sequence 的值。 */
	return Frame{Reply: packet(3, byte(device), byte(sequence)), CorrelationID: strconv.Itoa(sequence), State: map[string]any{"device": device, "sequence": sequence}}, nil /* 返回当前处理结果。 */
} /* 结束当前表达式或代码块。 */
