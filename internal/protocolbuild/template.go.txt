package main

import (
	"bufio"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"strings"
)

type RawMessage struct {
	Payload json.RawMessage `json:"payload"`
	PayloadFormat string `json:"payloadFormat"`
	DeviceID string `json:"deviceId"`
	Metadata map[string]any `json:"metadata"`
}

type Message struct {
	MessageType string `json:"messageType"`
	Properties map[string]any `json:"properties,omitempty"`
	Event map[string]any `json:"event,omitempty"`
}

// Decode 使用普通 Go 语法和标准库；可自由拆成多个函数或项目内的包。
// 示例报文 AA 01 2A：第三个字节表示温度，输出 temperature=42。
func Decode(raw RawMessage) (Message, error) {
	var text string
	if err := json.Unmarshal(raw.Payload, &text); err != nil {
		return Message{}, fmt.Errorf("报文应为十六进制字符串: %w", err)
	}
	data, err := hex.DecodeString(strings.Join(strings.Fields(text), ""))
	if err != nil { return Message{}, err }
	if len(data) != 3 || data[0] != 0xAA || data[1] != 0x01 {
		return Message{}, errors.New("报文须为 AA 01 加一个温度字节")
	}
	return Message{MessageType:"PROPERTY_REPORT", Properties:map[string]any{
		"temperature": int(data[2]),
	}}, nil
}

func respond(input []byte) map[string]any {
	var request struct { Version int `json:"version"`; Operation string `json:"operation"`; Raw RawMessage `json:"raw"` }
	err := json.Unmarshal(input, &request)
	if err == nil && (request.Version != 2 || request.Operation != "decode") { err = errors.New("expected go-protocol-v2 decode request") }
	var message Message
	if err == nil { message, err = Decode(request.Raw) }
	if err != nil { return map[string]any{"error": err.Error()} }
	return map[string]any{"standardMessage": message}
}

// main 是平台调用入口。标准输出只写解析结果，调试日志请写 os.Stderr。
// 平台以 IOT_PROTOCOL_WORKER_MODE=serve 常驻运行时逐行处理请求并带回 requestId；
// 请求之间不要依赖进程内存。
func main() {
	out := json.NewEncoder(os.Stdout)
	if os.Getenv("IOT_PROTOCOL_WORKER_MODE") == "serve" {
		scanner := bufio.NewScanner(os.Stdin)
		scanner.Buffer(make([]byte, 64<<10), 4<<20)
		for scanner.Scan() {
			var id struct { RequestID string `json:"requestId"` }
			_ = json.Unmarshal(scanner.Bytes(), &id)
			response := respond(scanner.Bytes())
			response["requestId"] = id.RequestID
			if out.Encode(response) != nil { os.Exit(1) }
		}
		return
	}
	var input json.RawMessage
	_ = json.NewDecoder(os.Stdin).Decode(&input)
	_ = out.Encode(respond(input))
}
